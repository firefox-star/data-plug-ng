    39→async function fetchWithTimeout(url: string, options: RequestInit = {}, timeout = 15000) {
    40→  const controller = new AbortController();
    41→  const id = setTimeout(() => controller.abort(), timeout);
    42→  try {
    43→    const res = await fetch(url, { ...options, signal: controller.signal });
    44→    clearTimeout(id);
    45→    return res;
    46→  } catch { clearTimeout(id); throw new Error('Request timed out'); }
    47→}
